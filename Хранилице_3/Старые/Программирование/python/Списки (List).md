

```
# Создание списка 
my_list = [1, 2, 3, 4, 5]

# Доступ к элементам 
print(my_list[0]) # Первый элемент: 1

# Изменение элемента 
my_list[1] = 10 # Теперь список: [1, 10, 3, 4, 5] 

# Добавление элементов 
my_list.append(6) # [1, 10, 3, 4, 5, 6] 

# Удаление элементов 
my_list.pop(1) # Удалить второй элемент, список станет [1, 3, 4, 5, 6] 

# Срезы 
print(my_list[1:4]) # [3, 4, 5] 

# Итерация 
for element in my_list: 
	print(element)

# Узнать длинну списка
len(my_list)

# Удалить конкретный элемент
my_list.remove(3) - Удалит элемент три
```
